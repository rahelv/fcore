% Two-tower vision-language model figure (image/text encoders -> shared space).
%
% Preamble requirements: \usepackage{tikz, amssymb}
%   \usetikzlibrary{arrows.meta,positioning,fit}
%
% Usage:
%   \begin{figure}[ht]
%       \centering

%       \caption{A vision-language model embeds images and texts into a
%       shared embedding space, where their similarity is measured by the
%       cosine of the angle between them.}
%       \label{fig:two_tower}
%   \end{figure}

\begin{tikzpicture}[
    >=Latex,
    box/.style={draw, thick, fill=gray!25, minimum width=3.0cm, minimum height=1.1cm, font=\normalsize, align=center},
    lbl/.style={font=\normalsize, align=center},
    emb/.style={circle, fill=black, inner sep=1.6pt},
%   emb/.style={circle, draw, thick, inner sep=1.5pt, font=\small, fill=white},
    flow/.style={->, thick, shorten <=6pt, shorten >=6pt}
]
% inputs
\node[inner sep=2pt]
    (img) at (0, 0) {\includegraphics[width=2.0cm]{spiderman-wondercon.jpg}};
\node[font=\small, below=2pt of img] {image $x$};
\node[minimum width=1.9cm, minimum height=1.5cm, align=center, font=\small] (txt) at (0, -3.0) {\textbf{``Spider Man} \\ \textbf{from Marvel''} \\ description $t$}; % TODO: evlt. add "pretrained" 

% encoders
\node[box] (enc1) at (4.2, 0)    {Image encoder\\ $f_{\text{img}}$};
\node[box] (enc2) at (4.2, -3.0) {Text encoder\\ $f_{\text{txt}}$};

\draw[flow] (img) -- (enc1);
\draw[flow] (txt) -- (enc2);

% coordinate system (shared embedding space)
\draw[->, very thick] (8.1, -3.2) -- ++(3.8, 0);
\draw[->, very thick] (8.6, -3.7) -- ++(0, 4);
\node[font=\normalsize, anchor=south] at (10.5, 0.3) {shared embedding space $\mathbb{R}^d$};

\node[emb] (u) at (9.5, -1.2) {};
\node[lbl, anchor=south west, font=\small] at (u.north east) {$\mathbf{u}$};
\node[emb] (v) at (10, -2.1) {};
\node[lbl, anchor=north west, font=\small] at (v.south east) {$\mathbf{v}$};

\draw[flow] (enc1) -- (u);
\draw[flow] (enc2) -- (v);

% similarity
\draw[thick, dashed] (u) -- (v) node[midway, right=3pt, font=\small] {$\cos(\mathbf{u}, \mathbf{v})$};
\end{tikzpicture}
